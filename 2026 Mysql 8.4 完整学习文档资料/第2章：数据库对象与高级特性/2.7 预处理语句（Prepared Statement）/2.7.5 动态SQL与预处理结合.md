# 动态 SQL 与预处理结合

动态 SQL 是指 SQL 语句的结构（表名、列名、WHERE 子句的组合等）在运行时根据条件决定，而不是在编写代码时固定。在 MySQL 存储过程中，通过将动态构建的 SQL 字符串赋给用户变量，然后使用 PREPARE + EXECUTE 执行，是实现动态 SQL 的标准模式。

---

## 基本动态 SQL 模式

```sql
DELIMITER //

-- 通用的动态分页查询
CREATE PROCEDURE sp_dynamic_page(
    IN  p_table  VARCHAR(64),
    IN  p_where  VARCHAR(500),
    IN  p_order  VARCHAR(100),
    IN  p_offset INT,
    IN  p_limit  INT
)
BEGIN
    -- 白名单校验：防止表名注入
    IF p_table NOT IN ('users', 'orders', 'products', 'articles') THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = '不允许访问该表';
    END IF;
    
    -- 排序方向白名单
    SET p_order = IF(p_order IN ('ASC', 'asc', 'DESC', 'desc'), UPPER(p_order), 'ASC');
    
    -- 动态构建 SQL
    SET @sql = CONCAT(
        'SELECT * FROM `', p_table, '` ',
        IF(p_where IS NOT NULL AND p_where != '', CONCAT('WHERE ', p_where, ' '), ''),
        'ORDER BY id ', p_order, ' ',
        'LIMIT ? OFFSET ?'
    );
    
    SET @p_limit  = p_limit;
    SET @p_offset = p_offset;
    
    PREPARE stmt FROM @sql;
    EXECUTE stmt USING @p_limit, @p_offset;
    DEALLOCATE PREPARE stmt;
END //

DELIMITER ;

CALL sp_dynamic_page('users', 'is_active = 1', 'DESC', 0, 10);
```

## 动态 ORDER BY 列名

```sql
DELIMITER //

-- 安全的动态排序（列名白名单校验）
CREATE PROCEDURE sp_users_sorted(
    IN p_sort_col VARCHAR(50),
    IN p_sort_dir VARCHAR(4),
    IN p_limit    INT
)
BEGIN
    -- 列名白名单（防止列名注入）
    SET p_sort_col = CASE p_sort_col
        WHEN 'username'   THEN 'username'
        WHEN 'email'      THEN 'email'
        WHEN 'created_at' THEN 'created_at'
        WHEN 'last_login' THEN 'last_login'
        ELSE 'id'    -- 默认按 id 排序
    END;
    
    -- 方向白名单
    SET p_sort_dir = IF(UPPER(p_sort_dir) = 'DESC', 'DESC', 'ASC');
    
    SET @sql = CONCAT(
        'SELECT id, username, email, created_at FROM users ',
        'ORDER BY `', p_sort_col, '` ', p_sort_dir, ' LIMIT ?'
    );
    SET @p_limit = p_limit;
    
    PREPARE stmt FROM @sql;
    EXECUTE stmt USING @p_limit;
    DEALLOCATE PREPARE stmt;
END //

DELIMITER ;

CALL sp_users_sorted('created_at', 'DESC', 20);
```

## 动态构建多条件查询

```sql
DELIMITER //

-- 根据传入的非 NULL 参数动态构建 WHERE 子句
CREATE PROCEDURE sp_search_orders(
    IN p_user_id   INT,
    IN p_status    TINYINT,
    IN p_start_dt  DATE,
    IN p_end_dt    DATE,
    IN p_min_amt   DECIMAL(10,2)
)
BEGIN
    SET @sql = 'SELECT id, user_id, amount, status, created_at FROM orders WHERE 1=1';
    
    IF p_user_id IS NOT NULL THEN
        SET @sql = CONCAT(@sql, ' AND user_id = ', p_user_id);
    END IF;
    
    IF p_status IS NOT NULL THEN
        SET @sql = CONCAT(@sql, ' AND status = ', p_status);
    END IF;
    
    IF p_start_dt IS NOT NULL THEN
        SET @sql = CONCAT(@sql, ' AND created_at >= ''', DATE_FORMAT(p_start_dt, '%Y-%m-%d'), '''');
    END IF;
    
    IF p_end_dt IS NOT NULL THEN
        SET @sql = CONCAT(@sql, ' AND created_at <= ''', DATE_FORMAT(p_end_dt, '%Y-%m-%d'), ' 23:59:59''');
    END IF;
    
    IF p_min_amt IS NOT NULL THEN
        SET @sql = CONCAT(@sql, ' AND amount >= ', p_min_amt);
    END IF;
    
    SET @sql = CONCAT(@sql, ' ORDER BY created_at DESC LIMIT 100');
    
    PREPARE stmt FROM @sql;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;
END //

DELIMITER ;

-- 只传部分参数（其他条件不生效）
CALL sp_search_orders(5, NULL, '2026-01-01', NULL, 100.00);
-- 生成：WHERE 1=1 AND user_id = 5 AND created_at >= '2026-01-01' AND amount >= 100
```

## 动态 PIVOT（行转列）

```sql
DELIMITER //

-- 动态列名的 PIVOT 查询（按月份生成列）
CREATE PROCEDURE sp_monthly_pivot(IN p_year INT)
BEGIN
    DECLARE v_cols VARCHAR(2000) DEFAULT '';
    DECLARE i INT DEFAULT 1;
    
    -- 动态生成列定义（12 个月）
    WHILE i <= 12 DO
        SET v_cols = CONCAT(
            v_cols,
            IF(i > 1, ', ', ''),
            'SUM(CASE WHEN MONTH(created_at) = ', i, 
            ' THEN amount ELSE 0 END) AS `month_', LPAD(i, 2, '0'), '`'
        );
        SET i = i + 1;
    END WHILE;
    
    SET @sql = CONCAT(
        'SELECT user_id, ', v_cols,
        ' FROM orders WHERE YEAR(created_at) = ', p_year,
        ' GROUP BY user_id ORDER BY user_id'
    );
    
    PREPARE stmt FROM @sql;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;
END //

DELIMITER ;

CALL sp_monthly_pivot(2026);
```

## 动态 DDL 操作

```sql
DELIMITER //

-- 动态创建按月分区的表（如日志表）
CREATE PROCEDURE sp_create_monthly_log_table(IN p_yearmonth CHAR(6))
BEGIN
    -- 校验格式（YYYYMM）
    IF p_yearmonth NOT REGEXP '^[0-9]{6}$' THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = '年月格式必须为 YYYYMM';
    END IF;
    
    SET @table_name = CONCAT('logs_', p_yearmonth);
    SET @sql = CONCAT(
        'CREATE TABLE IF NOT EXISTS `', @table_name, '` (',
        '  `id` BIGINT AUTO_INCREMENT PRIMARY KEY,',
        '  `user_id` INT,',
        '  `action` VARCHAR(100),',
        '  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,',
        '  INDEX `idx_user_id` (`user_id`)',
        ') ENGINE=InnoDB DEFAULT CHARSET=utf8mb4'
    );
    
    PREPARE stmt FROM @sql;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;
    
    SELECT CONCAT('创建表 ', @table_name, ' 成功') AS result;
END //

DELIMITER ;

CALL sp_create_monthly_log_table('202603');
```

## 适用场景

- **可选过滤条件**：根据用户传入的参数动态决定 WHERE 子句的组合
- **动态排序**：支持前端传入排序列和方向（必须白名单校验）
- **分表查询**：动态选择按月/按年分表的表名执行查询
- **数据透视（PIVOT）**：动态生成列名实现行转列报表

## 常见问题

**动态 SQL 中的字符串参数如何防注入？**

如果值通过 `?` 占位符传入（`EXECUTE stmt USING @val`），MySQL 自动处理，安全。如果需要将字符串值直接拼接进 SQL（如日期字符串），应使用 `DATE_FORMAT()` 或严格的格式校验确保不含危险字符，并用单引号包裹。对于整数类型（用户ID、状态码），拼接前确保是整数类型（`CAST(@p AS SIGNED)`）。

---

## 注意事项

- 动态 SQL 中的标识符（表名、列名）无法用 `?` 参数化，必须通过白名单（`CASE WHEN` 或 `IF IN` 校验）确保安全。切勿将用户输入直接拼接为表名或列名。
- 动态 SQL 的调试比静态 SQL 困难。建议在存储过程中加入调试模式：`IF @debug THEN SELECT @sql AS debug_sql; END IF;`，方便排查拼接出的 SQL 是否正确。

---

## 总结

动态 SQL 通过字符串拼接构建 SQL 文本，存入用户变量后用 PREPARE + EXECUTE 执行。常用模式：`SET @sql = CONCAT(...); PREPARE s FROM @sql; EXECUTE s [USING @params]; DEALLOCATE PREPARE s;`。值参数用 `?` 传入（安全），结构参数（表名/列名）必须白名单校验后再拼接。动态 SQL 是存储过程实现通用查询框架、分表路由的核心工具。
